---
name: brownfield-code-change
description: >-
  既存のリポジトリをユーザーの依頼に従って変更するときに使う。
  ソフトウェアコードと、エージェントが実行するスキルの変更を含む。
  設計相談、調査、レビューなどリポジトリを読み取るだけの依頼には使わない。
---

```python
def condition(prompt: str) -> bool: ...
def constraint(prompt: str): ...
def context(prompt: str): ...
def do(prompt: str): ...


def 既存のコード・スキルの変更():
    with (
        context("""
            対象は、既存の利用者と設計を持つソフトウェアとスキルとする。
            スキルはエージェントが読む指示として扱い、本文、参照資料、補助プログラムと、それを読み込んで実行する経路を含める。
        """),
        constraint("""
            この作業の完了は、依頼された結果が実際に成立したところまでとする。
            スキルで設計するか、実装中やレビュー後に設計へ戻るかは、入力と成果物の約束、情報や記録の表現と管理、判断・操作を担う主体、開始・終了・委譲・受け渡しの条件と、入出力・データの表現・状態の所有・処理の責務の対応によって決まる。
            現在のコードや指示にそう書かれているという理由だけで、その状態を保持すべき仕様とはみなさない。
            作業中にユーザーが依頼を訂正または変更した場合は、すでに作成した `要件契約` を書き直し、以後の判断も現在有効な内容へ合わせる。
            設計上の選択はユーザーの依頼や既存の仕様ではないので、新しい要求として `要件契約` へ追加しない。
            依頼とは独立した問題の修正、根拠のない互換動作や代替経路、ついでの再設計は持ち込まない。
        """),
    ):
        変更する対象を確定する()
        依頼を後続へ保持する()
        if condition("""
            依頼された結果と守るべき条件を、現在有効なユーザー発言と確認した対象から確定できた。
            一般知識や慣例で不足を補った部分は、確定したことにしない。
        """):
            with constraint("`software-design` を適用するかを判定できない場合は、適用するものとして扱う。"):
                if condition("今回の変更が、`software-design` の `description` が示す適用条件に当たる"):
                    with constraint("既存の設計から決まっている部分を新しく決め直さない。"):
                        do("本体のコード、スキルの本文・参照資料・補助プログラムを編集する前に、今回の変更とその接続先について `software-design` に従って設計を作り、独立した確認を終える。")
            with constraint("""
                実装方法の選択によって利用者が受け取る結果や既存の契約が変わる場合、その選択が依頼・仕様・既存設計から決まらなければ勝手に確定しない。
                変更前から存在する利用者の編集を、自分の変更と混同して上書きしたり取り消したりしない。
                作業記録は対象の規則に従い、既存のものを使うか、なければ用意する。
                依頼を保持する `要件契約` とは分けて扱う。
                指示どおりに経路をたどった確認と、エージェントが実際に実行した結果を区別し、前者を後者の証拠にはしない。
                構文検査やコマンドの正常終了だけで、求められた動作まで成立したことにはしない。
                実装を正しいことにするために、期待結果や検査条件を変更しない。
            """):
                変更を作る()
                実際の結果を確認する()
                レビューして完了する()
        else:
            do("実装へ進まず、確認できたことと、何が決まらないため変更できないかをユーザーへ返す。")


def 変更する対象を確定する():
    do("編集を始める前に、依頼が実際のどのコード・指示と利用経路に対応するかを確認する。")
    if not condition("対象だけで挙動が分かる"):
        do("参照先や利用側までたどり、適用される仕様や既存の検証も確認する。")
    do("この確認で得た事実に合わせて対象理解を直す。")


def 依頼を後続へ保持する():
    with constraint("ユーザー発言とその参照先だけで確認できる依頼には、それ以上の意味を追加しない。"):
        do("現在有効なユーザー発言を、後続の作業者が直接参照できる `要件契約` として固定する。")
    if condition("探索によって、後続の行動を拘束する内容が新たに確定した"):
        do("その内容だけについて `requirement-contract` に従って候補を作り、その追加部分を `requirement-reviewer` と `review-orchestration` で独立に確認してから採用する。")


def 変更を作る():
    do("編集する対象の作業開始時の内容を、後から参照できることを編集前に確かめる。")
    if condition("編集する対象に、履歴から復元できない未コミット・未追跡の内容がある"):
        do("その内容を、編集前に作業記録へ残す。")
    if condition("`software-design` で設計を作った"):
        do("確認した設計、利用例、変更例、確認結果を、別の実装計画や要件へ書き写さず、そのまま実装に使う。")
    do("依頼された結果に必要な変更を、既存の責務分担と利用経路に沿って行う。")
    if condition("変更によって、既存の対応関係が崩れる"):
        do("認可された変更を成立させるために必要な追従も揃える。")
    if condition("実装中の変更が、確認した入出力の約束、データの表現、状態の所有、処理の責務を変える"):
        do("変更後の設計に依存する確認を、`software-design` に従って行う。")


def 実際の結果を確認する():
    with constraint("同じ対象と条件ですでに得た有効な結果は再利用し、同じ確認を繰り返すためだけに実行しない。"):
        do("対象に用意されているビルド等の検証方法を使い、依頼された結果を判定できる方法で確認する。")
        if condition("対象がスキル"):
            do("適用される指示の品質基準へ照合し、実際の読み込み経路と実行者が使える情報から、判断・操作・受け渡しと結果を確認する。")
        if condition("不具合を修正した"):
            do("問題の成立を確認した条件を、変更後にも適用する。")
        if condition("確認の結果に、想定と違ったものがある"):
            do("その結果に合わせて対象理解と実装を見直し、見直した後の対象を、依頼された結果を判定できる方法で確認する。")


def レビューして完了する():
    with constraint("修正案を採れなかったことを、元の欠陥が解消したことにしない。"):
        with constraint("""
            今回の依頼の成立に関わらず、修正の影響も受けない箇所から、さらに改善候補を探し続けることは必須工程にしない。
            `software-design` で作った設計に照らして、前提とする入力の値や状態に当たるものが無かった指摘では、その入力の扱いを、実装の修正として決めない。
        """):
            do("実装とその検証を終えた後の、変更後のコードやスキル本文とその参照先を、レビューの対象にする。")
            while not condition("レビューの対象を最後に決めた後に、その対象をレビューした"):
                with constraint("`software-design` で確認した設計を書き換えない。"):
                    with constraint("""
                        `software-design` で確認した設計は、レビュアーには渡さない。
                        レビューでは今回の依頼を示す `要件契約` と、対象の要求を定める既存の正本をそれぞれ直接参照させる。
                    """):
                        do("レビューの対象を `artifact-review` と `review-orchestration` でレビューする。")
                    for 指摘 in その回の指摘:
                        if condition("`software-design` で設計を作った"):
                            do("その指摘が前提とする条件を、設計が定めた入力の範囲とその保証に照らす。")
                            if condition("その指摘が前提とする条件に、設計が定めた入力の範囲とその保証で当たるものがある"):
                                do("添えられた出どころを直接開いて確かめる。")
                            do("照らした結果と開いた出どころを、その指摘の成立と対応を確かめた根拠に含める。")
                        do("その指摘の成立を確かめる。")
                    今回の対応を決める()

                if (
                    condition("`software-design` で設計を作った")
                    and condition("指摘が前提とする入力の値や状態で、当たるものが設計に無いものを、この作業で扱う")
                ):
                    do("`software-design` に従って、その入力の範囲とその保証を設計で決める。")

                if condition("今回解消すると決めた問題が残っている"):
                    直し方を比べて変更する()
                    if condition("修正が入った"):
                        do("修正前後の差分も、指摘や対応を選んだ理由と同じ作業記録に残す。")
                        do("元の問題が成立した条件と、修正の影響を受ける利用経路・既存の約束について検証する。")
                        with constraint("""
                            前の回から対象・条件が変わらず、修正の影響も受けない検証・レビュー結果は、やり直しの対象にしない。
                            影響先を確認せずに差分の行だけへ絞ったり、影響しない部分の全面レビューを繰り返したりしない。
                        """):
                            do("元の問題が成立した条件と、修正の影響を受ける利用経路・既存の約束を、次の独立レビューでも確かめる。")
                            with constraint("次のレビューの対象には、修正の差分だけでなく、変更した定義の利用側と、入出力や状態の変化が伝わる先も含める。"):
                                do("次のレビューの対象を決め直す。")

        with constraint("""
            ユーザーの指定を見直す材料となる影響も、指定に従ったという理由で報告から落とさない。
            修正の費用が大きいことだけを理由に、必要な品質を満たさないまま完了にしない。
        """):
            do("現在有効な依頼へ戻り、求められた結果と守るべき性質が成立し、必要な追従・検証・レビューが終わり、今回解消すべき問題が残っていないかを確かめる。")
            do("必要な確認が `要調査`・`判定不能` のまま残っていないか、依頼の成立に必要なユーザーの選択が残っていないかも確かめる。")
            if condition("未解消の指摘がある"):
                do("成立条件と影響、今回修正しない根拠となった指定や要求との関係を最終報告に示す。")
            if (
                condition("求められた結果と守るべき性質が成立していることを確かめた")
                and condition("必要な追従・検証・レビューが終わったことを確かめた")
                and condition("""
                    今回解消すべき問題が残っていないことを確かめた。
                    すべての指摘が消えたかどうかでは判定しない。
                """)
                and condition("必要な確認に、`要調査`・`判定不能` のまま残っているものが無いことを確かめた")
                and condition("依頼の成立に必要なユーザーの選択が、残っていないことを確かめた")
            ):
                do("完了とする。")
            else:
                do("依頼された結果を確認できないまま作業を終える場合は、現在成立している状態、未解決の問題、続行できない理由と必要な判断を返す。")


def 今回の対応を決める():
    if condition("成立した指摘がある"):
        with constraint("""
            修正を始めない。
            設計者だけが選んだ非保証は、ユーザーの指定と同じ権威を持たない。
            設計に書かれていることだけを免除の根拠にせず、指摘が成立したことだけを撤回の根拠にもしない。
        """):
            do("前の回までに作業記録に残した指摘、確認した根拠、対応を選んだ理由を、すべて、今回の対応を決めるのに使う。")
            今回の依頼との関係を確かめる()
            修正の経過から原因を確かめる()
            with constraint("""
                成立したことだけで修正義務が生じるわけではなく、指摘に添えられた改善案も変更の認可にはならない。
                「高品質に」のような一般的な望みだけを、既存の独立した問題をすべて直す根拠にしない。
            """):
                do("今回の扱いと、解消する問題をまとめて直す範囲を決める。")
    with context("履歴から再取得できるものは参照でよい。"):
        do("指摘、確認した根拠、対応を選んだ理由を、後の編集で失われない形で作業記録に残す。")


def 今回の依頼との関係を確かめる():
    with constraint("""
        判断に必要な事実は、取得できる資料や実物で確認したものだけを使う。
        未確認のことを「以前からある」「影響が小さい」「対象外」として処理しない。
        判断待ちの事項を、依存しない作業を止める理由にしない。
    """):
        do("ユーザーの選択が必要なものと、自分で調べられる不足を分ける。")
        for 指摘 in 成立した指摘:
            do("その指摘について、失われる動作や性質、発生条件、その条件へ到達する利用経路を確認する。")
            if condition("その指摘が保守上の問題"):
                do("実際の定義や更新処理をたどり、一つの変更に対して誰がどこを揃えなければならないかを示す。")
            with constraint("""
                仮想の機能追加から対応義務を作らない。
                設計者が選んだ実現方法を要求として固定しない。
            """):
                do("現在の誤動作と、変更時に発生する作業や漏れの危険を区別する。")
                do("その影響を、今回求められた結果、適用仕様、既存の利用側が依存する約束、成果物に適用される品質基準へ照合する。")
                do("何が成立しなくなるかと、その約束の出どころを対応付ける。")

            if condition("ユーザーが明示した非対応・非保証が、その指摘の条件に関わる"):
                with constraint("""
                    同じ条件で別の性質まで失われる場合は、それも許容されたとはみなさない。
                    例えば、狭い画面で全文を表示しない指定は、アプリケーションの停止まで許す指定ではない。
                """):
                    do("元の発言を読み、対象の条件と許容された結果の範囲を確かめる。")
                    if condition("その指摘の影響に、許容された範囲に収まるものがある"):
                        do("その範囲に収まる影響は、欠陥が成立していても、その指定を覆して修正しない。")
                    if condition("その指摘の影響に、許容された範囲に収まらず、今回保つべき性質が失われるものがある"):
                        if condition("指定や外部仕様を変えずに、その性質を回復できる"):
                            do("その性質の回復を、今回解消する問題に含める。")
                        if condition("その性質の回復には、指定や外部仕様の変更が要ることを確かめた"):
                            do("その変更を実行せず、元の指定、判明した影響、選択によって変わる結果をユーザーへ返す。")

            if condition("設計者だけが選んだ非保証が、その指摘の条件に関わる"):
                do("その判断が前提にした入力、利用経路、影響を指摘の事実と比べる。")
                if condition("その判断の前提が崩れた箇所がある"):
                    do("前提が崩れた箇所と失われる性質を設計の検討へ戻す。")

            if (
                condition("その指摘が、今回の変更で持ち込んだ欠陥である")
                or condition("その指摘が、依頼された結果や守るべき約束の成立を妨げる欠陥である")
            ):
                do("以上の指定と認可を守り、その欠陥を、この作業で解消する問題に含める。")

            do("作業開始時の内容と今回の変更の影響先を確認し、同じ問題が変更前にもあり、今回の変更から独立していて、今回の要求を妨げないかを確かめる。")
            if (
                condition("同じ問題が変更前にもあったことを確かめた")
                and condition("その指摘が、今回の変更から独立した欠陥であることを確かめた")
                and condition("その指摘が今回の要求を妨げないことを確かめた")
            ):
                do("修正範囲に加えず、成立条件と影響を報告する。")


def 修正の経過から原因を確かめる():
    with constraint("現在のコードや指示だけから発生時期を推測しない。"):
        with constraint("作業開始時にあった利用者の未コミット変更も比較の基準に含める。"):
            do("成立した指摘全体について、今回の成果物、作業開始時の内容、前の回までの修正前後を、指摘が成立する同じ条件で比べる。")
        do("既存の記録や差分で確認できる事実は再利用し、因果関係が決まらない箇所をコードや指示の追跡、実行で確かめる。")
    do("各問題について、結果を変える値、その値の所有者と更新元、依存する処理をたどる。")
    if condition("対象がスキル"):
        do("判断に使う情報や記録、その取得・更新を担う主体、依存する指示と受け渡しをたどる。")
    with constraint("同じファイルにあることや、修正後に指摘されたことだけを共通原因や因果関係の根拠にしない。"):
        do("複数の指摘が同じデータ表現、更新規則、責務分担に由来するか、前の修正で加えた状態、同期、分岐、呼び出し順が今回の問題を生んだかを確かめる。")
    if condition("以前からある独立した欠陥を新しく発見した"):
        do("その欠陥を、修正方法が次の欠陥を生んだ場合と分ける。")
    if condition("前の修正が新しい問題を生んだ"):
        do("前に解消しようとした問題と今回の問題を並べ、両方を生む値の持ち方や処理の約束を検討対象にする。")
    if condition("複数の問題が同じ判断に由来する"):
        do("その判断を一緒に扱う。")


def 直し方を比べて変更する():
    with constraint("""
        指摘された箇所へ手を加える案を既定路線にしない。
        分岐が増えること自体を品質低下とはしない。
        一件の指摘を消しても、別の約束を破る案は採らない。
        前の修正が新しい問題を生んだ場合は、前の修正を維持することを前提に補正を足さない。
        既存の約束に従って誤記や計算を正せば解消する独立した問題まで、一括して再設計しない。
        レビューの回数、指摘数、変更行数から、設計へ戻るかどうかを決めない。
    """):
        do("今回解消する問題の直し方の候補について、問題を解消する仕組みと、増減する状態、分岐、同期、呼び出し順、変更時の追従箇所を、修正前と同じ利用・変更の場面で比べる。")
        do("追加または残す負担には、それを除くと失われる動作や約束を対応付ける。")
        do("確認した原因と守る条件から、直し方を選ぶ。")
        if condition("""
            同じ要求と既存の約束を保ち、同期や追従の仕事を減らせる案がある。
            別の場所へ同じ手動同期を移しただけなら、同期を減らしたことにはしない。
        """):
            do("その案を、比較結果を根拠に選ぶ。")
        if condition("負担の交換がある"):
            do("何が減り何が増えるかを示し、今回の目的と既存の方針に照らして選ぶ。")

        if condition("利用者が受け取る結果や認可範囲を変える選択が残っている"):
            do("その選択を勝手に確定せず、ユーザーへ返す。")
            with constraint("その選択に依存する変更は、ユーザーの判断を待つ。"):
                do("その選択に依存しない変更を進める。")

        if condition("選んだ案がデータ表現、状態の所有、入出力の約束、処理の責務を変える"):
            with constraint("""
                関係しない設計工程までやり直さない。
```
