---
name: mechanism-textbook
description: >-
  複雑な対象を、読者が仕組みや推論のつながりから深く理解できるようにする教科書や体系的な解説を作るときに使う。
  対象には、機構や実装とその背景にある理論、複数の層にまたがる系の依存とデータの流れ、観測データから主張を導く分析の筋道を含む。
  対象を説明できる知識体系を再構成し、読者の既有知識から理解を積み上げられる順序と、文章・図表・式・模型による表現で教える。模型の形は一次資料と照合し、説明の設計は理解の条件を満たすかを確かめ、本文は設計と根拠に合うかを確かめる。
  個別の答えを返せば足りる質問、既知の項目を引くためだけの参照資料、結果や経過を伝えるだけで読者が推論をたどる必要のない報告の作成には使わない。

                    while (

                            if condition("保留を解いた項目に、その選択で、今回直す指摘が無くなったものがある"):
                                do("その項目を、作業記録で、確認済みにする。")
                        if not condition("根拠と形の仮定の工程が、確認済みである"):
                            根拠と形の仮定をそろえて確かめさせる()
                        if condition("根拠と形の仮定の工程が、確認済みである") and not condition("設計の工程が、確認済みである"):

                        if condition("設計までの工程が、どれも確認済みである") and not condition("章の図と章の本文の工程が、どれも確認済みである"):

                        if condition("章の本文までの工程が、どれも確認済みである") and not condition("本全体の工程が、確認済みである"):

def 理解の到達点と出発点を確かめる():
    do("依頼から、読者が何をどこまで理解したいのかを確かめる。")
    if condition("理解した後の利用目的がある"):
        do("その利用目的を、説明範囲と深さの判断に使う。")
    # 「この式を説明できる」のような書き方だけでは、読者に何を思い描かせるかが決まらず、図を描く空間も決まらない。
    # だから、読者が頭の中に組み立てられるようになるものも、到達点に書く。
    # たとえば、光線がシーンの中をどう進んで反射するか、制御信号がどの層をどの周期で受け渡されるか、二つの計算方法がそれぞれどの点から何を求めるか。
    with constraint("""
        到達点は、答えを掲載する質問の一覧ではなく、読者が説明できるようになる性質や関係として書く。
        読者が頭の中に組み立てられるようになるものも書く。
    """):
        do("到達点を書く。")
    do("その理解に必要な、依頼で名指しされていない背景知識も扱う。")
    # どの水準を目指すかによって、説明の順序と確かめ方が変わる。だから、求める水準を確かめておく。
    with constraint("用語や結果を思い出せること、仕組みを説明できること、本文に無い場面で結果を導けることは、別の水準として扱う。"):
        do("依頼から、読者に求める推論の水準を確かめる。")
    with constraint("名前や一般的な役割を知っていることと、内部の仕組みを理解していることは区別する。"):
        do("読者がすでに知っていることを確かめる。")
    do("使える資料の所在と確認方法、成果物の形式、読者がそれを読む場所（ブラウザ、課題管理システムの本文、エディタなど）、途中で人間の評価を受けられるかを確認する。")
    if condition("資料を調べても決まらない、構成を左右する不明点がある"):
        with constraint("資料から分かることは、人間に聞き直さない。"):
            do("その不明点だけを質問する。")
    do("質問しなかった不明点について置いた仮定を、明示する。")
    if condition("対象に版や構成の違いがある"):
        do("主に扱う範囲と比較として触れる範囲を決める。")
    do("依頼の原文、到達点、推論の水準、読む場所、このスキルの所在を、作業記録に残す。")
    with context("範囲を文書で読める既有知識（教育課程の科目、特定の教科書など）は、その文書の所在と範囲（科目、章など）で記録してよい。"):
        do("読者の知識について分かったことは、出どころ（読者本人や依頼者の発言、記録）とともに残す。")


def 対象を説明できる知識体系を再構成する():
    if condition("調査や照合を担当に分ける"):
        do("担当に、作業に必要な対象理解に加え、その説明の位置付け、読者がそこまでに学んだこと、到達すべき理解と話の筋を渡す。")
    with constraint("""
        資料に明記されたこと、実装から確認したこと、実行・実測で確認したこと、根拠からの推論、未確認のことを区別する。
        一般論と対象固有の事実を混同しない。
        解消できない資料の食い違いを、無理に一つの説明へまとめない。
    """):
        do("資料を調べ、対象の重要な性質が、何を前提に、どのような関係によって成り立つかを明らかにする。")
        do("個々の事実だけでなく、それらを結んで結果を導く筋道を確かめる。")
        do("背景知識は、信頼できる資料から補う。")
        if condition("対象が、観測データから主張を導く分析である"):
            do("データが何をどのように観測したものか、処理の各段、処理の結果から主張へ進む推論の規則を、前提と関係として明らかにする。")
        if condition("資料に十分な説明がある"):
            do("その説明を利用する。")
        if (
            condition("資料の説明が分散している")
            or condition("資料の説明が欠けている")
        ):
            do("根拠へ戻って組み立て直す。")
        do("既存の実測結果は、条件を確認して利用する。")
        do("追加の実行・測定が必要な点だけを確かめる。")
        if condition("資料が食い違う"):
            do("対象や条件の違いを確かめる。")
        with (
            context("""
                まとまりは物理的な部品とは限らない。
                必要なつながりを保つことと、内部の細部をすべて収集することは別である。
            """),
            constraint("特定の質問への答えに要る部分だけを切り出して、同じ仕組みを断片化しない。"),
        ):
            do("知識を、一つの概念や仕組みを理解できるまとまりとして捉える。")
            do("概念の区別、数量の関係、複数の要素に共通する原理など、対象に合う単位を選ぶ。")
            do("まとまりの範囲を、到達点に書いた性質を説明するために必要な前提と関係から決める。")
        # この種類によって、結果を導いて見せる進め方と、表現の形が変わる。だから、まとまりごとに見分けておく。
        with context("種類は、何がどこにあり何とつながるか（構造）、時間とともに状態がどう変わるか（振る舞い）、形・位置・方向・距離の関係（空間と形）、定義から結果を導く式（式と導出）、データから主張までの推論のつながりなどである。"):
            do("まとまりごとに、読者が理解するときに頭の中に組み立てるものの種類を見分け、作業記録に残す。")
        # 対応物がある事柄は、読者がひとまとまりとして扱える。無い事柄は、読者がその構成を一つずつ保持しなければならない。
        # だから、有無を確かめておき、章の順序と説明の深さを決めるときに使う。
        do("まとまりごとに、そのまとまりで教える事柄について、読者の既有知識の記録の中に対応物があるかを確かめる。")
        do("対応物の有無と、あればそれが何かを、作業記録に残す。")
        do("資料の記述を言い換えるだけでは答えられない例について、再構成した前提と関係から結果と理由を導き、対象に応じて、実装、実測、定義や論証などと照合する。")
        if condition("入力と規則から結果を計算で再現できる対象である"):
            do("再構成した前提と関係を、模型にする。")
            do("模型の出力を、模型とは別に得た値（実装を実行した結果、試験、実測、公開された値、元データの一部を別の方法で数え直した結果など）と照合し、照合した値と条件を作業記録に残す。")
        if (
            condition("導けない例がある")
            or condition("結果が合わない例がある")
        ):
            do("知識の欠落だけでなく、前提、規則、推論、比較条件の誤りも調べる。")
        do("再構成した前提と関係、および模型が置いている形の仮定を、一つずつ作業記録に書き出す。")
        do("形の仮定に、それを述べている一次資料（規格、論文、仕様、実装、実測の記録など）の箇所を、出典として付ける。")
        if condition("出典が見つからない形の仮定がある"):
            出典の無い仮定を扱う()
        with constraint("形の仮定から外した仮定を前提にした模型の出力は、使わない。"):
            do("対象理解について、範囲、根拠と適用条件、未解明の点、模型と照合の結果、形の仮定とその出典を、作業記録に残す。")
            if condition("根拠が足りない部分がある"):
                do("根拠で説明できる範囲を、作業記録に残す。")


def 出典の無い仮定を扱う():
    if condition("出典が見つからない仮定がある"):
        do("資料を探す。")
        if condition("探しても出典が見つからない仮定がある"):
            do("その仮定を形の仮定から外して、作業記録で区別する。")
            do("その仮定と、それに拠る模型の出力、図、導出の段を、根拠に支えられないものとして扱う。")


# 模型の値を照合した相手が、同じ形の仮定から計算した値（式による計算など）だと、形を取り違えていても値は一致する。
# だから、値の照合とは別に、形の仮定そのものを一次資料と照らさせる。
def 根拠と形の仮定をそろえて確かめさせる():
    if condition("根拠、模型、形の仮定に、直すと決めたものがある"):
        根拠と形の仮定を直す()
    if condition("形の仮定に、保留でも確認済みでもないものがある"):
        with constraint("項目は、形の仮定のうち、保留でも確認済みでもないものの一つ一つとする。"):
            do("確かめる項目を決めて、作業記録に残す。")
        with constraint("渡すものと渡さないもの、返させるものは、確認の手順の資料（references/review.md）の「形の仮定の確認」のとおりにする。"):
            do("確認者への依頼を用意する。")
            do("依頼の中で、同じ資料の「確認者に共通の規則」を守らせる。")
        確認者に依頼する()
        指摘を扱う()


def 根拠と形の仮定を直す():
    if (
        condition("直すと決めたものに、根拠に支えられない主張がある")
        or condition("直すと決めたものに、出典の無い導出がある")
    ):
        do("「対象を説明できる知識体系を再構成する」に戻って、資料、実行、測定で確かめる。")
        if condition("確かめても根拠を得られなかった主張や導出がある"):

    if (
        condition("直すと決めたものに、形の仮定がある")
        or condition("直すと決めたものに、模型がある")
    ):

    出典の無い仮定を扱う()
    if condition("直すと決めたものに、模型がある"):

    if condition("模型を作った対象である") and not condition("模型の照合の記録が作業記録にある"):

    if (
        condition("形の仮定を直した")
        or condition("模型を直した")

# 書き手は、図や本文が何を示すはずかを知っているので、描かれていないものや書かれていない前提を、あるものとして読んでしまう。
# だから、確かめる対象を作っていない者に確かめさせる。
# 項目を決めずに全体から欠陥を探させると、確認者ごとに別の箇所を見つけ、直すたびに新しい指摘が出て、確認が終わらない。
# だから、先に決めた項目だけを確かめさせる。
def 確認者に依頼する():

    """):
        if condition("サブエージェントを起動できる"):

                do("用意した依頼ごとに、確かめる対象を作っていない、この確認のために新しく起動したサブエージェントを確認者にし、その依頼を渡して、項目を確かめさせる。")
        else:
            do("編集責任者が、同じ条件と項目の一覧を使って、自分で確かめる。")
            do("独立した確認を行っていないことを、作業記録に残し、報告でも伝える。")


# 原因が上流の工程にある指摘は、上流から直すので、先に下流を直しても作り直しになる。
# だから、ここでは、原因のある工程と直し方を決めて残すだけにし、直すのは、その工程を次に進めるときにする。
def 指摘を扱う():
    指摘 = define("確認者が返した箇所と、編集責任者が照合で見つけた箇所のうち、満たさない条件に当たるとされたもの。")
    成り立つ指摘 = define("指摘のうち、編集責任者が対象と根拠に照らして、満たさない条件に当たると確かめたもの。")
    do("その回の確認で確かめた項目と結果を、後の対応判断に使えるよう作業記録に残す。")
    with constraint("未確認を、成り立たないことや解消したことにしない。"):
        指摘が成り立つかを確かめる(指摘, 成り立つ指摘)
        with (
            context("今回の到達点と品質の条件の成立に関わらず、修正の影響も受けない箇所から、新たな改善候補を探し続けることは必須にしない。"),
            constraint("既存の設計に従って直せる独立した誤記まで、一括して設計し直さない。"),
            constraint("""
                成り立つことだけで、指摘された箇所を直す方法や、扱う範囲を広げることを決めない。
                回数によって削除や完了を決めない。
            """),
        ):
            今回の対応を選ぶ(指摘, 成り立つ指摘)
        if condition("確認者が、確認用スクリプトで画像を作れないために確かめられない条件（成果物の動作など）だけを確かめられなかった項目がある"):
            do("その項目は、確かめていない条件を作業記録に残して、確かめられた項目として扱う。")
        if condition("ほかに、確認できず進められない項目がある"):
            do("その項目を、作業記録で保留にする。")
        if condition("依頼者の選択を待つ指摘がある"):
            do("その指摘の項目を、依頼者の返事が届くまで、作業記録で保留にする。")
        if (
            condition("必要な資料や環境を得られず確認できないことがある")
            or condition("到達点や依頼者の指定の変更が必要である")
        ):
            do("残る問題、影響、続行に必要な情報や判断を報告する。")
        if condition("根拠を得られず扱えない部分が残っている"):
            do("未達の範囲と理由を報告する。")
    do("確認者が確かめられた項目のうち、保留にしておらず、その項目についての指摘に、今回直すと決めたものも、成り立つかを確かめられなかったものも無い項目を、作業記録で、確認済みにする。")


def 指摘が成り立つかを確かめる(指摘, 成り立つ指摘):
    # 書き手の意図は、指摘を退ける理由にならない。問われているのは、図や本文がその意図を読者に伝えているかだから。
    with constraint("""
        同じ条件と問題を述べる重複した指摘は、一つとして扱う。
        満たさない条件に当たらない返答は、指摘として扱わない。

        書き手の意図を理由に指摘を退けない。
    """):
        do("その回の確認がすべて戻ってから、指摘ごとに対象と根拠の資料を直接開き、確認の手順の資料の「観点と、満たさない条件」に照らして、満たさない条件に当たるか、どれに当たるかと、その理由を作業記録に残す。")
        do("当たると確かめた指摘を、成り立つ指摘として残す。")
    if condition("確かめられないことがある"):
        do("取得できる資料や実物を調べる。")
        if condition("調べても確かめられないことが残っている"):
            do("確かめられない範囲と理由を残す。")


def 今回の対応を選ぶ(指摘, 成り立つ指摘):
    with constraint("""
        編集責任者の選択を、修正の免除や新たな要求の根拠にしない。
        成り立つ問題を、費用が大きいことや確認回数だけで対象外にしない。
        変更前からある独立した問題として、今回の範囲に加えない候補は、確かめられたものだけを、今回の範囲に加えない。
    """):
        do("成り立つ指摘が、どの条件で、到達点、章を読み終えた読者が何を理解できるようになるか、導出、または四つの品質の条件のどれを妨げるかを、依頼の原文と作業記録に照らす。")
        do("読者の既有知識、対象の範囲、推論の水準についての依頼者の指定と、編集責任者が選んだ章立て、説明位置、図の分担を区別する。")
        do("今回の到達点と適用する品質の条件の成立を妨げる問題、および今回の修正が持ち込んだ問題を、解消する対象にする。")
        if condition("依頼者が明示した非対応の範囲がある"):
            do("その範囲を元の発言で確かめ、その指定が許した影響と、同じ条件で別の性質が失われることを分ける。")
        if condition("依頼者の指定を変えなければ解消できない指摘がある"):
            do("その変更を実行せず、指定、影響、必要な選択を依頼者へ返す。")
        if condition("変更前からある独立した問題として、今回の範囲に加えない候補がある"):
            do("作業開始時の内容と影響先で、同じ問題が前からあり、今回の到達点と品質の条件を妨げず、修正の影響も受けていないことを確かめる。")

        for 指摘 in 成り立つ指摘:
            do("その指摘で問題になった事柄の、生成元と利用先をたどる。")
            if condition("その指摘が、形の仮定についての指摘である"):
                do("一次資料と、再構成した前提と関係、模型を照合する。")
            if condition("その指摘が、説明に関わる指摘である"):
                do("意味を教える節、初めて使う位置、章の前提関係、導出の段、図と本文の分担をたどる。")
        with constraint("同じ章にあることや、修正後に指摘されたことだけを、共通原因や因果関係の根拠にしない。"):
            do("複数の指摘が同じ仮定、説明位置や前提関係に由来するか、前の修正で加えた説明や参照が別の未説明の事柄や矛盾を生んだかを確かめる。")
            do("以前からある独立した問題の後発発見は、前の直し方が問題を生んだ場合と分ける。")
        if condition("前の修正が問題を生んだ"):
            do("前に解消しようとした問題と今回の問題を並べ、両方を生む説明位置や前提関係を検討する。")
        do("原因と守る条件から、その箇所を直す、教える位置や章の前提関係を変える、導出や図と本文の分担を組み直す、到達点に不要な部分を外す、といった案を挙げる。")
        if condition("形の仮定を直す案がある"):
            do("その案の修正前後を、一次資料の記述に照らして比べる。")
            if condition("模型を作った対象である"):
                do("その照合結果も使う。")
        if condition("章の設計や本文を直す案がある"):
            do("その案の修正前後に、同じ確認用の例と導出の段を当て、欠けた前提が揃うか、別の未説明の前提が生じないか、本文の決まりのうち章と節の書き方を保てるかを確かめる。")
        do("案ごとに、説明の重複、参照の行き来、本文と図で別々に揃える情報の増減を、修正前と同じ条件で比べる。")

        if condition("部分を外す案がある"):

                do("負担の交換がある案は、到達点と既存の方針に照らして選ぶ。")
        if condition("設計を変える案を選んだ節に、ほかの成り立つ指摘で、設計に原因があり、今回の範囲に加えないとも、依頼者へ返すとも決めていないものがある"):
            do("その指摘も、設計で一緒に直すものに選ぶ。")
        do("指摘、確かめた根拠、対応を選んだ理由を、後の編集で失われない形で作業記録に残す。")
        do("今回直すと決めたものごとに、原因のある工程の何を、どう直すかを、作業記録の、直すと決めたものに残す。")


# 本文は設計を文章にするだけで、書く段階では、推論も前提も主張も足さない。設計に足りないものは、本文を書いても埋まらない。
# だから、図と本文を作る前のこの確認で、設計だけから本文を書けることを確かめさせる。
def 設計をそろえて確かめさせる():
    if not condition("設計を作ってある"):
        読者が理解を獲得する順序を設計する()
    if condition("設計に、直すと決めたものか、作った後に、作る元にしたものや拠るものが変わった項目がある"):
        設計を直す()
    if condition("根拠と形の仮定の工程が、確認済みである"):
        with constraint("項目は、到達点と、各章の前提にする章、節ごとの記録、対応物の記録、図の計画、確認用の例、導出の段の一つ一つのうち、保留でも確認済みでもないものとする。"):
            do("確かめる項目を決めて、作業記録に残す。")
        with context("章ごとに確認者を分けて並行して依頼してよい。"):
            with constraint("渡すものと渡さないもの、見る満たさない条件、返させるものは、確認の手順の資料の「設計の確認」のとおりにする。"):
                do("確認者への依頼を用意する。")
                do("依頼の中で、同じ資料の「確認者に共通の規則」を守らせる。")
            if condition("章ごとに確認者を分けて依頼する"):
                do("章の順序の条件（読み進めやすさ 13）は、すべての章の設計を一人の確認者に照合させる依頼にする。")
            確認者に依頼する()
        指摘を扱う()


def 読者が理解を獲得する順序を設計する():
    with constraint("設計は、根拠で説明できる範囲で組む。"):
        do("表現の資料（references/representation.md）と、本文の決まりの資料（references/writing.md）の「目的と原則」を含む全体を読む。")
        do("読者が読む場所と成果物の形式から、媒体を、表現の資料の「媒体を決める」で決める。")
        with (
            context("""
                読者が一度に頭の中に保持できる情報には限りがある。
                理解の負荷は、項目の数よりも、同時に関係づけて保持しなければならない要素の多さで決まる。
            """),
            constraint("情報を種類で分けるのではなく、説明の中で何を支えるかを見る。"),
        ):
            章の順序と重みを決める()
            with constraint("対象が分析の場合、処理の結果から主張へ進む推論も、節で述べる推論に含める。"):
                for 章 in 本の章:
                    章の設計を記録する(章)
        # 本の説明だけで到達点の推論ができるかを、書き手と確認者が確かめられるように、確認用の例を作業記録に置く。
        do("到達点に書いた推論ごとに、到達点と再構成した知識体系が決める範囲から、確認用の例を選ぶ。")
        do("確認用の例ごとに、本文のどの節まで読んだ読者が導けるはずかを添える。")
        for 推論 in 確認用の例の結果と理由を本の内容から導く推論:
            導出の段を書く(推論)
        do("確認用の例とその導出の段を、作業記録に置く。")
        説明せずに使う事柄を読者に確かめる()
        do("知識の前提関係、用語の意味と教える節、章ごとの話の筋を、作業記録に残す。")


def 章の順序と重みを決める():
    with constraint("用語の定義だけでなく、次の説明を理解するために使う考え方や関係も前提に含める。"):
        do("知識のまとまりごとに、理解に必要な前提をたどり、読者の既有知識までつなぐ。")
    # 読者は本文を少しずつ読み進め、新しく読む部分を、その時点で頭に残っている少数の事柄にしか結び付けられない。どの事柄が頭に残っているかは、書き手には予測できない。
    # 長い説明の読者は、必要な章から読み始め、分からない所があれば前の章へ戻る。
    # だから、読者が何を覚えているかを予測しなくて済むように、章を読み始める単位と決めて、章の順序と前提を決める。
    with (
        context("""
            読者は、章から読み始める。
            ある章が別の章を前提にするとは、その章で読者に結果を導かせる箇所が、別の章で教える事柄の理解を使うことである。
            そこで知っていれば足りることだけを述べて使う事柄は、前提にしたことにならない。
            対象の一通りの動きを先に体験させる入口の章や、それまでの章で学んだことを組み合わせて使う章を置いてもよい。
        """),
        constraint("""
            ある章が前提にする事柄を教える章は、その章より前に置く。
            互いに依存しない章どうしは、教える事柄のうち、対応物が読者の既有知識にあるものの割合が大きい章から先に並べる。
            入口の章は、後の章を前提にしない。
        """),
    ):
        with constraint("""
            順序は、読者が一度に保持できる情報の限りに合わせて選ぶ。
            対象の構造、資料の構成、読者が学ぶ順序を同一視しない。
            調査の分担や資料の目次を、そのまま章立てにしない。
        """):
            do("この依存関係を、読者が学べる順序へ組み直し、章の順序を決める。")
            do("下位層からの説明、具体例からの一般化、概略からの詳細化など、前提の飛躍を避けられる進め方を選ぶ。")
        do("章ごとに、前提にする章と、読み終えた読者が何を理解できるようになるかを決める。")
    with constraint("重みは、手元にある素材の量ではなく、到達点への寄与で決める。"):
        do("章ごとに、到達点にとっての重みを決め、重みに応じて紙幅と詳しさを配分する。")
    do("決めた章の順序、前提にする章、読み終えた読者が理解できるようになること、重みを、作業記録に残す。")


def 章の設計を記録する(章):
    do("依頼の原文、その章に関わる記録、この手順の「読者が理解を獲得する順序を設計する」、本文の決まりの資料、表現の資料を読み直す。")
    # 依存関係を並べただけでは、話の筋にはならない。だから、章ごとに、読者がたどる筋を組み立てる。
    with constraint("各章を独立した事実集にせず、前の説明で得た理解が次の足場になるようにする。"):
        do("その章で、読者が何を知っているところから始め、何を見て考え、どの理解へ到達するのかを組み立てる。")
        do("新しい知識が、前後の説明にどうつながるかを明らかにする。")
        if condition("最初は概略を教え、後で理解を深める構成を使う"):
            do("初めの説明が成り立つ範囲を保ち、後で深める知識を先に使わない組み立てにする。")
        # 読者がすでに知っている事柄を説明し直すと、その読者には余分な説明になり、学習を妨げることがある。
        # だから、対応物がある事柄は、違いだけを説明する。
        for 事柄 in その章で教える事柄:
            if condition("その事柄に対応物が、読者の既有知識にある"):
                do("その事柄は、対応物を示し、違いを説明することにする。")
